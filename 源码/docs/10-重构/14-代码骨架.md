```python
# app/ai/model/client.py
from dataclasses import dataclass
import httpx, time

@dataclass
class ModelReply:
    content: str
    model: str
    prompt_tokens: int
    completion_tokens: int
    latency_ms: int

class ModelError(Exception): ...
class ModelFormatError(ModelError): ...

class ModelClient:
    def __init__(self, api_key: str, base_url: str = "https://api.deepseek.com"):
        self._key = api_key
        self._base = base_url.rstrip("/")
        self._http = httpx.Client(timeout=60)

    def complete(self, messages, model, *, max_tokens=800,
                 temperature=0.2, json_mode=False) -> ModelReply:
        body = {
            "model": model,
            "messages": messages,
            "max_tokens": max_tokens,
            "temperature": temperature,
        }
        if json_mode:
            body["response_format"] = {"type": "json_object"}

        last = None
        for attempt in range(3):
            began = time.monotonic()
            try:
                r = self._http.post(
                    f"{self._base}/chat/completions",
                    headers={"Authorization": f"Bearer {self._key}"},
                    json=body,
                )
                if r.status_code == 429 or r.status_code >= 500:
                    raise RuntimeError(f"HTTP {r.status_code}")
                r.raise_for_status()
                data = r.json()
                usage = data.get("usage", {})
                return ModelReply(
                    content=data["choices"][0]["message"]["content"],
                    model=data.get("model", model),
                    prompt_tokens=usage.get("prompt_tokens", 0),
                    completion_tokens=usage.get("completion_tokens", 0),
                    latency_ms=int((time.monotonic() - began) * 1000),
                )
            except Exception as e:  # noqa: BLE001
                last = e
                time.sleep(1.5 ** attempt)
        raise ModelError(str(last))
```

```python
# app/ai/model/schema.py
import json, re
from pydantic import BaseModel, ValidationError

def parse_json(text: str):
    text = text.strip()
    text = re.sub(r"^```(?:json)?|```$", "", text, flags=re.M).strip()
    return json.loads(text)

def structured(client, messages, model, schema: type[BaseModel], **kw):
    reply = client.complete(messages, model, json_mode=True, **kw)
    try:
        return schema.model_validate(parse_json(reply.content)), reply
    except (json.JSONDecodeError, ValidationError) as e:
        repair = messages + [
            {"role": "assistant", "content": reply.content},
            {"role": "user", "content": f"输出不合法：{e}。只输出修复后的 JSON。"},
        ]
        reply2 = client.complete(repair, model, json_mode=True, **kw)
        try:
            return schema.model_validate(parse_json(reply2.content)), reply2
        except Exception as e2:  # noqa: BLE001
            raise ModelFormatError(str(e2)) from e2
```

```python
# app/ai/cost/ledger.py
def record_cost(db, *, run_id, stage, model, reply) -> float:
    price = price_of(model, now_ms())
    cost = (reply.prompt_tokens * price.in_price
            + reply.completion_tokens * price.out_price) / 1_000_000
    db.add(CostRecord(
        run_id=run_id, stage=stage, model=model,
        prompt_tokens=reply.prompt_tokens,
        completion_tokens=reply.completion_tokens,
        price_tier=price.tier, cost_cny=cost,
        latency_ms=reply.latency_ms, created_at=now_ms(),
    ))
    run = db.get(Run, run_id)
    run.total_cost_cny = float(run.total_cost_cny or 0) + cost
    db.commit()
    return cost
```

```python
STAGES = ["plan", "design", "build", "assemble", "test", "deliver"]

def run(run_id: str) -> None:
    try:
        for stage in STAGES:
            if halted(run_id):
                return
            emit_phase(run_id, stage)
            ctx = load_ctx(run_id)
            tasks = planner.plan(run_id, stage, ctx)
            if not tasks:
                fail(run_id, f"{stage} 没有产出任务")
                return
            ok = scheduler.run_stage(run_id, stage, tasks, ctx)
            if not ok and stage == "test":
                if not loopback.fix_and_retest(run_id, ctx):
                    fail(run_id, "修复后仍未通过")
                    return
                continue
            if not ok:
                fail(run_id, f"{stage} 失败")
                return
            checkpoint.save(run_id, stage, ctx)
            if stage == "plan" and ctx.spec.questions:
                gate.wait(run_id, ctx.spec)
        set_status(run_id, "done")
        emit_phase(run_id, "done")
    except BudgetExceeded:
        set_status(run_id, "paused")
        emit_error(run_id, "预算到顶")
    except Exception as e:  # noqa: BLE001
        emit_error(run_id, str(e))
        set_status(run_id, "failed")
```

## 5. 调度器

```python
# app/pipeline/orchestrator/scheduler.py
def run_stage(run_id, stage, tasks, ctx) -> bool:
    executor = ThreadPoolExecutor(max_workers=CONCURRENCY)
    done, failed = set(), []
    futures = {}

    def submit(task):
        futures[executor.submit(execute, run_id, task, ctx)] = task

    pending = list(tasks)
    while pending or futures:
        for task in list(pending):
            if all(d in done for d in task.depends_on):
                pending.remove(task)
                emit_task(run_id, "created", task)
                submit(task)

        if not futures:
            # 还有 pending 但依赖永远不满足 → 依赖成环
            return False

        for fut in concurrent.futures.as_completed(list(futures)):
            task = futures.pop(fut)
            try:
                result = fut.result()
                done.add(task.id)
                emit_task(run_id, "done", task, result=result)
            except TaskFailed as e:
                failed.append(task)
                emit_task(run_id, "failed", task, reason=e.reason)
                if task.attempts < 2:
                    task.attempts += 1
                    pending.append(task)   # 自动重试一次
    executor.shutdown()
    return not failed
```

```python
# app/tools/files.py
def _safe(root: Path, path: str) -> Path:
    target = (root / path).resolve()
    if not str(target).startswith(str(root.resolve())):
        raise PermissionError("路径越界")
    return target

def write_file(root, path, content):
    target = _safe(Path(root), path)
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_text(content, encoding="utf-8")
```

```python
# app/tools/shell.py
ALLOWED = {"python", "python3", "ls", "dir", "cat", "type"}

def run_command(cmd: list[str], timeout=30):
    if cmd[0] not in ALLOWED:
        raise PermissionError("命令不在白名单")
    proc = subprocess.run(cmd, capture_output=True, timeout=timeout, shell=False)
    return proc.returncode, proc.stdout.decode()[:8000], proc.stderr.decode()[:2000]
```

```python
def start(run_id, files_dir, entry, port):
    env = {**os.environ, "PORT": str(port)}
    proc = subprocess.Popen(
        [sys.executable, entry], cwd=files_dir, env=env,
        stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL,
    )
    if not wait_http(f"http://127.0.0.1:{port}/", timeout=15):
        proc.terminate()
        raise RuntimeError("预览启动超时")
    return proc
```

```python
def emit(run_id, kind, payload):

        last = db.execute(select(func.max(RunEvent.seq)).where(RunEvent.run_id == run_id)).scalar() or 0
```
