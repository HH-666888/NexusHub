`showGate`：

```js
function showGate(gate) {
  const box = document.getElementById("gate-modal");
  if (!box) return;
  const qs = gate.questions || [];
  box.querySelector(".gate-body").innerHTML = qs.map(function (q, i) {
    return '<div class="gate-q"><p>' + (i + 1) + ". " + q.text + "</p>" +
      '<div class="gate-opts">' + (q.options || []).map(function (o) {
        return '<label><input type="radio" name="' + q.id + '" value="' + o + '"> ' + o + "</label>";
      }).join("") + "</div></div>";
  }).join("");
  box.hidden = false;
  box.dataset.run = id;
}

async function submitGate(auto) {
  const box = document.getElementById("gate-modal");
  const answers = [];
  box.querySelectorAll(".gate-q").forEach(function (block) {
    const checked = block.querySelector("input:checked");
    const name = block.querySelector("input").name;
    answers.push({ id: name, value: checked ? checked.value : "" });
  });
  await window.api.answerGate(id, { answers: answers, auto: Boolean(auto) });
  box.hidden = true;
}
```

`describeResult`：

```js
function describeResult(r) {
  var parts = [];
  if (r.files) parts.push(r.files.length + " 个文件");
  if (r.lines) parts.push(r.lines + " 行");
  if (r.checks) parts.push(r.checks + " 项检查");
  return parts.join(" · ") || "—";
}
```

### 1.3 任务定位（新增）

侧栏点击已有事件委托，在 `showTaskDetail` 之后加：

```js
function locateTask(taskId) {
  var task = TASK_STORE.tasks.find(function (t) { return t.id === taskId; });
  if (!task) return;
  var index = STAGE_ORDER.indexOf(task.stage);
  if (index < 0) return;
  if (view.focus !== null && view.focus !== index) view.focus = null;
  view.expanded.add(index);          // 展开折叠
  render();
  requestAnimationFrame(function () {
    var node = document.querySelector('[data-task="' + taskId + '"]');
    if (!node) return;
    node.scrollIntoView({ behavior: "smooth", block: "center" });
    node.classList.add("is-flash");
    setTimeout(function () { node.classList.remove("is-flash"); }, 700);
  });
}
```

CSS：

```css
.node-task.is-flash { box-shadow: 0 0 0 3px var(--primary-soft), 0 10px 24px rgba(0,0,0,.10); }
```
